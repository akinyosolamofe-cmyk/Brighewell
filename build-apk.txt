      - name: Wrap the website as an Android app
        working-directory: app
        run: |
          mkdir www && cp -r bridgewell/* www/
          npm init -y
          npm install @capacitor/core@6 @capacitor/cli@6 @capacitor/android@6
          npx cap init "Bridgewell" com.bridgewell.app --web-dir www
          npx cap add android
          npx cap sync android
      - name: Build the APK
        working-directory: app/android
        run: ./gradlew assembleDebug

          name: bridgewell-apk
